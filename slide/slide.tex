\begin{frame}{Qu'est-ce qu'une PRF ?}

\textbf{PRF = Pseudorandom Function}
(fonction pseudo-aléatoire)

\vspace{0.5cm}

C'est une fonction déterministe

\[
F_k : \{0,1\}^n \rightarrow \{0,1\}^m
\]

paramétrée par une clé secrète \(k\)

\vspace{0.5cm}

Sans connaître \(k\), \(F_k(x)\) est indistinguable d'une fonction aléatoire

\end{frame}

%------------------------------------------------
\begin{frame}{Contexte}
\begin{itemize}
    \item Les PRF sont utilisées en cryptographie moderne :
    \begin{itemize}
        \item chiffrement
        \item authentification
        \item dérivation de clés
    \end{itemize}

    \vspace{0.3cm}

    \item Problème :
    comment déléguer seulement une partie des capacités d'évaluation ?

    \vspace{0.3cm}

    \item Solution :
    les \textbf{Constrained Pseudorandom Functions (CPRF)}.
\end{itemize}
\end{frame}


%-----------------------------------------------
\begin{frame}{Principe des CPRF}

Une CPRF est une PRF dont l'évaluation peut être restreinte

\vspace{0.4cm}

Exemple de contrainte :

\[
C_n(x) = [x < n]
\]

\vspace{0.4cm}

\begin{itemize}
    \item Clé maître : accès complet
    \item Clé contrainte : accès limité à certaines entrées
\end{itemize}

\vspace{0.5cm}

Objectif :
déléguer des capacités sans révéler la clé secrète
\end{frame}

\begin{frame}{Utilisation de RSA}

Permutation publique :
\[
\pi(x) = x^e \bmod N
\]

Inverse de la permutation :
\[
\pi^{-1}(x) = x^d \bmod N
\]

\vspace{0.4cm}

\begin{itemize}
    \item La clé privée permet de remonter la chaîne
    \item La clé publique permet uniquement d'avancer
    \item RSA fournit une permutation bijective adaptée à la construction
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Clé contrainte}

Pour une contrainte \(x < n\), on donne :
\[
(PK,\ ST_n,\ n) \qquad \text{avec } ST_n = \pi^{-n}(ST_0)
\]

L'évaluation contrainte devient :
\[
\mathit{EvalC}(c) = \pi^{n-c}(ST_n)
\]

\begin{itemize}
    \item Possible si \(c < n\).
    \item Impossible d'aller au-delà sans la clé privée
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{Attaque CJ25}

L'attaquant obtient :
\begin{itemize}
    \item une clé contrainte \((PK, ST_n, n)\)
    \item des évaluations hors contrainte \(\mathit{Eval}(x)\) pour \(x > n\)
\end{itemize}

\vspace{0.4cm}

Puis il utilise la structure de la chaîne de permutation

\begin{frame}{Alternative : permutation AES}

\textbf{Idée} : remplacer RSA par AES comme permutation à trappe

\vspace{0.4cm}

AES définit une bijection sur les blocs de 128 bits :
\[
f_k(x) = E_k(x), \qquad f_k^{-1}(x) = D_k(x)
\]

\vspace{0.4cm}

\textbf{Problème fondamental :}

\begin{itemize}
    \item Pour inverser, il faut \textbf{obligatoirement} connaître la clé $k$.
    \item Il est impossible de donner une clé contrainte \textbf{sans révéler} la clé $k$.
\end{itemize}

\vspace{0.3cm}


\end{frame}

%------------------------------------------------
\begin{frame}{Alternative : fonction de Rabin}

La fonction de Rabin est définie par :
\[
f(x) = x^2 \bmod n, \qquad n = p \cdot q
\]

\vspace{0.3cm}

\textbf{Inversion} (avec la clé secrète $(p, q)$) :
résoudre $x^2 \equiv y \pmod{n}$ donne \textbf{4 racines} :
\[
\pm r_p \cdot q \cdot (q^{-1} \bmod p) \;\pm\; r_q \cdot p \cdot (p^{-1} \bmod q) \pmod{n}
\]

\vspace{0.3cm}

\textbf{Problème fondamental :}
\begin{itemize}
    \item $f$ n'est \textbf{pas injective} : une image admet 4 pré-images.
    \item Même avec la clé privée, on ne sait pas laquelle est la \og bonne \fg.
    \item Il est donc impossible d'itérer $f^{-1}$ de façon déterministe.
\end{itemize}

\vspace{0.3cm}


\end{frame}

\begin{frame}{Alternative : $F^{\mathrm{WEAK}}$ --- construction}

\textbf{Idée} : utiliser une application linéaire comme permutation.

\vspace{0.3cm}

\begin{columns}[t]
\begin{column}{0.48\textwidth}
\textbf{Clé maître}
\[
S \in (\mathbb{Z}/p\mathbb{Z})^{M \times N}
\]
matrice choisie aléatoirement.

\vspace{0.3cm}

\textbf{Évaluation}
\[
\mathit{Eval}(S, \vec{x}) = S \cdot \vec{x}
\]
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Clé contrainte pour $\vec{y}$}

L'oracle tire $\vec{d} \neq \vec{0}$ et renvoie :
\[
S_{\vec{y}} = S + \vec{d} \cdot \vec{y}^T
\]

\vspace{0.3cm}

\textbf{Évaluation contrainte}
\[
\mathit{EvalC}(S_{\vec{y}}, \vec{x}) = S_{\vec{y}} \cdot \vec{x}
\]
correcte si et seulement si $\langle \vec{x}, \vec{y} \rangle = 0$.
\end{column}
\end{columns}

\end{frame}

%------------------------------------------------
\begin{frame}{Attaque sur $F^{\mathrm{WEAK}}$ --- principe}

\textbf{Objectif} : retrouver la clé maître $S$ à partir d'une clé contrainte et d'évaluations.

\vspace{0.3cm}

\textbf{Choix de l'attaquant} : demander la clé contrainte pour
\[
\vec{y} = (0, \ldots, 0, 1)
\]

Il reçoit $S_{\vec{y}} = S + \vec{d} \cdot \vec{y}^T$.

\vspace{0.3cm}

\textbf{Observation clé} : pour tout $\vec{x} \perp \vec{y}$,
\[
S_{\vec{y}} \cdot \vec{x} = S \cdot \vec{x}
\]
Les $N-1$ premières colonnes de $S$ sont directement accessibles via $S_{\vec{y}}$.

\vspace{0.3cm}

Il reste à retrouver la dernière colonne $S_{\cdot, N}$.

\end{frame}

%------------------------------------------------
\begin{frame}{Attaque sur $F^{\mathrm{WEAK}}$ --- résolution}

\textbf{Pour retrouver $S_{i,N}$} : demander $\mathit{Eval}(\vec{x})$ avec $x_N \neq 0$.

Pour chaque ligne $i$ de $S$, on a :
\[
S_{i,1} x_1 + \cdots + S_{i,N-1} x_{N-1} + S_{i,N} x_N = y_i
\]

Puisque $S_{i,1}, \ldots, S_{i,N-1}$ sont connus, on en déduit :
\[
S_{i,N} = x_N^{-1}\left(y_i - \sum_{j=1}^{N-1} S_{i,j} x_j\right)
\]


\vspace{0.3cm}

\begin{itemize}
    \item Si l'oracle utilise une fonction aléatoire : l'attaquant reçoit une valeur aléatoire de $S_{i,N}$
    \item Si l'oracle utilise $F^{\mathrm{WEAK}}$ : l'attaquant reçoit la bonne valeur de $S_{i,N}$
    \item Après quelques requêtes, l'attaquant retrouve que la bonne valeur $S_{i,N}$ est celle qui revient le plus souvent
\end{itemize}



\end{frame}

%------------------------------------------------
\begin{frame}{Renforcement de $F^{\mathrm{WEAK}}$ par hachage}

\textbf{Idée} : composer $F^{\mathrm{WEAK}}$ avec une fonction de hachage $H$.

\vspace{0.3cm}

\begin{columns}[t]
\begin{column}{0.48\textwidth}
\textbf{Sans hachage}
\[
\mathit{Eval}(\vec{x}) = S \cdot \vec{x}
\]
\textcolor{red}{$\Rightarrow$ attaque linéaire possible.}

\vspace{0.4cm}

\textbf{Avec hachage}
\[
\mathit{Eval}_H(\vec{x}) = H(S \cdot \vec{x})
\]
Retrouver $S$ nécessite une pré-image de $H$.
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Théorème (CJ25)}
Soit $CF$ une CPRF. Si :
\begin{enumerate}
    \item $|\mathit{CF.Out}(\lambda,x)|$ est super-polynomial en $\lambda$,
    \item $CF$ est IND$^*$-NC-CPRF sécurisée,
    \item l'oracle aléatoire est non-programmable,
\end{enumerate}
alors $\mathsf{Hashed}[CF]$ est\\
\textbf{SIM$^*$-AC-CPRF sécurisée}.
\end{block}
\end{column}
\end{columns}

\end{frame}

%------------------------------------------------
\begin{frame}{Vérification --- Condition 1 : super-polynomial}

\textbf{Condition 1 : } $|\mathit{F^{\mathrm{WEAK}}.Out}(\lambda, x)|$ est super-polynomial en $\lambda$.

\vspace{0.3cm}

Les coefficients sont dans $\mathbb{Z}/p\mathbb{Z}$ avec $|p| \approx \lambda$ bits.\\
La matrice $S$ contient $M \times N$ coefficients, donc :
\[
|\mathit{F^{\mathrm{WEAK}}.Out}| \approx 2^{\lambda \cdot M \cdot N}
\]
\textcolor{green!60!black}{\checkmark} \textbf{Condition 1 vérifiée.}

\end{frame}

%------------------------------------------------
\begin{frame}{Vérification --- Condition 3 : oracle non-programmable}
 
\textbf{Condition 3 :} l'oracle aléatoire $H$ est \textbf{non-programmable}.
 
\vspace{0.3cm}



\textbf{Implémentation d'une fonction de hachage par lazy sampling :}
 
\vspace{0.2cm}


\begin{block}{Définition}
Un oracle $P$ est non-programmable si :
\begin{itemize}
    \item on ne peut pas \textbf{forcer} $P(z) = y$ pour une valeur $y$ choisie à l'avance,
    \item la distribution de $P(z)$ pour une entrée non encore évaluée est toujours \textbf{uniforme et indépendante}.
\end{itemize}
\end{block}
 
\vspace{0.3cm}
 
\begin{itemize}
    \item Si $z$ \textbf{déjà évalué} : retourner la valeur mémorisée $H(z)$.
    \item Sinon : tirer $y \leftarrow \{0,1\}^\lambda$ aléatoirement, mémoriser $(z, y)$, retourner $y$.
\end{itemize}
 
\vspace{0.15cm}

La valeur de $H(z)$ est fixée \textbf{au moment de la première évaluation} et ne peut plus être modifiée.
 
\vspace{0.3cm}
 
\textcolor{green!60!black}{\Large\checkmark} \textbf{Condition 3 vérifiée.}
 
\end{frame}
 
%------------------------------------------------
\begin{frame}{Vérification --- Condition 2 : IND$^*$-NC-CPRF (1/2)}
 
\textbf{Condition 2 :} $F^{\mathrm{WEAK}}$ est IND$^*$-NC-CPRF sécurisée.
 
\vspace{0.3cm}
 
\begin{block}{Rappel : que signifie IND$^*$-NC ?}
La CPRF $CF$ est IND$^*$-NC sécurisée si \textbf{aucun attaquant efficace} ne peut :
\begin{enumerate}
    \item demander \textbf{toutes} ses évaluations d'un coup (non-adaptativement),
    \item effectue ses calculs,
    \item renvoye sa décision à l'oracle tout en ayant un \textbf{avantage significatif}
\end{enumerate}
\end{block}
 
\vspace{0.3cm}
 
Pour prouver que $F^{\mathrm{WEAK}}$ n'est \textbf{pas} IND$^*$-NC sécurisée, il suffit de décrire un tel attaquant.
 
\end{frame}
 
%------------------------------------------------
\begin{frame}{Vérification --- Condition 2 : IND$^*$-NC-CPRF (2/2)}
 
\textbf{Jeu de sécurité simplifié contre $F^{\mathrm{WEAK}}$ :}
 
\vspace{0.1cm}
 
\begin{enumerate}
 
    \item[\textbf{1.}] \textbf{Clé contrainte.} $\mathcal{A}$ demande $S_{\vec{y}}$ pour $\vec{y} = (0,\ldots,0,1)$.\\
    $\Rightarrow$ $\mathcal{A}$ récupère les $N-1$ premières colonnes de $S$.
 
 
    \item[\textbf{2.}] \textbf{Évaluations (non-adaptatives).} $\mathcal{A}$ envoie \textbf{en une seule fois}\\
    $n$ vecteurs $\vec{x}^{(1)}, \ldots, \vec{x}^{(n)}$ avec $x^{(k)}_N \neq 0$,\\
    et reçoit toutes les réponses $\vec{y}^{(1)}, \ldots, \vec{y}^{(n)}$ en retour.
 
 
    \item[\textbf{3.}] \textbf{Calcul.} Pour chaque requête $k$ et chaque ligne $i$, $\mathcal{A}$ calcule :
    \[
    \tilde{S}_{i,N}^{(k)} = \bigl(x^{(k)}_N\bigr)^{-1}\!\left(y^{(k)}_i - \sum_{j=1}^{N-1} S_{i,j}\, x^{(k)}_j\right)
    \]
 
 
    \item[\textbf{4.}] \textbf{Décision.} Pour les $\tilde{S}_{i,N}^{(k)}$ \textbf{identiques} $\Rightarrow$ CPRF.\\
    Sinon $\Rightarrow$ fonction aléatoire.
\end{enumerate}
 
\vspace{0.1cm}
 
\textcolor{red!60!black}{$\times$} \textbf{Condition 2 non vérifiée.}
 
\end{frame}
 
%------------------------------------------------
\begin{frame}{Conclusion sur $\mathsf{Hashed}[F^{\mathrm{WEAK}}]$}
 

\begin{center}
\begin{tabular}{clc}
\hline
\textbf{Condition} & \textbf{Énoncé} & \textbf{Statut} \\
\hline
1 & Espace de sortie super-polynomial & \textcolor{green!60!black}{\checkmark} \\
2 & IND$^*$-NC-CPRF sécurisée & \textcolor{red}{$\times$} \\
3 & Oracle non-programmable (lazy sampling) & \textcolor{green!60!black}{\checkmark} \\
\hline
\end{tabular}
\end{center}
 
\vspace{0.4cm}
 
La condition 2 n'est pas vérifiée : on ne peut pas utiliser ce théorème pour
prouver la sécurité de $\mathsf{Hashed}[F^{\mathrm{WEAK}}]$.

 
\vspace{0.3cm}
 

Cela \textbf{ne prouve pas} que $\mathsf{Hashed}[F^{\mathrm{WEAK}}]$ est non sécurisée.\\[0.2cm]
La sécurité de $\mathsf{Hashed}[F^{\mathrm{WEAK}}]$ reste une \textbf{question ouverte}.

 
\end{frame}
